cmake_minimum_required(VERSION 3.25)
project(rumblex_perception_face)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(rumblex_interfaces REQUIRED)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc calib3d objdetect dnn)
add_executable(node_face_detector src/node_face_detector.cpp)
target_include_directories(node_face_detector PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(node_face_detector rclcpp::rclcpp cv_bridge::cv_bridge
  ${sensor_msgs_TARGETS} ${std_msgs_TARGETS} ${rumblex_interfaces_TARGETS} ${OpenCV_LIBS})
install(TARGETS node_face_detector DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config models DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_face_tracking test/test_face_tracking.cpp)
  target_include_directories(test_face_tracking PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(test_face_tracking ${OpenCV_LIBS})
  target_compile_definitions(test_face_tracking PRIVATE FACE_MODEL_PATH="${CMAKE_CURRENT_SOURCE_DIR}/models/face_detection_yunet_2023mar.onnx")
endif()
ament_package()
